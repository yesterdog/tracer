Mutation = Callable[[Path], None]
MUTANTS: dict[str, tuple[str, Mutation]] = {}


def mutant(name: str, text: str) -> Callable[[Mutation], Mutation]:
    def register(fn: Mutation) -> Mutation:
        MUTANTS[name] = (text, fn)
        return fn

    return register


def edit(root: Path, rel: str, old: str, new: str, count: int = 1) -> None:
    path = root / rel
    data = path.read_text(encoding="utf-8")
    assert old in data, (rel, old[:60])
    path.write_text(data.replace(old, new, count), encoding="utf-8")


def read(root: Path, rel: str) -> str:
    return (root / rel).read_text(encoding="utf-8")


def write(root: Path, rel: str, data: str) -> None:
    (root / rel).write_text(data, encoding="utf-8")

def run(name: str, quiet: bool = True) -> tuple[bool, list[str]]:
    text, fn = MUTANTS[name]
    if WORK.exists():
        shutil.rmtree(WORK)
    shutil.copytree(BASE, WORK)
    fn(WORK)
    env = {
        **os.environ,
        "PYTHONPATH": str(WORK / "packages/evalschema/src"),
        "PYTHONDONTWRITEBYTECODE": "1",
    }
    proc = subprocess.run(  # noqa: S603
        [PY, "-m", "pytest", "-q", "--tb=no", "-p", "no:cacheprovider", "-x" if False else "-q",
         "-rf", "packages/evalschema/tests/test_golden_fixtures.py"],
        cwd=WORK,
        env=env,
        capture_output=True,
        text=True,
        timeout=600,
    )
    failed = [line for line in proc.stdout.splitlines() if line.startswith("FAILED")]
    errors = [line for line in proc.stdout.splitlines() if line.startswith("ERROR")]
    killed = proc.returncode != 0
    return killed, (failed + errors)


if __name__ == "__main__":
    names = sys.argv[1:] or list(MUTANTS)
    survivors = []
    for name in names:
        killed, lines = run(name)
        text = MUTANTS[name][0]
        verdict = "KILLED  " if killed else "SURVIVED"
        print(f"{name} {verdict} {text}")
        for line in lines[:4]:
            print("     ", re.sub(r"packages/evalschema/tests/test_golden_fixtures.py::", "", line)[:200])
        if len(lines) > 4:
            print(f"      ... {len(lines)} failing in total")
        if not killed:
            survivors.append(name)
        sys.stdout.flush()
    print("SURVIVORS:", survivors)
